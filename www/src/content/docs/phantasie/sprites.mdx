import SpriteGallery from '../../../components/SpriteGallery.astro';
import Lightbox from '../../../components/Lightbox.astro';
import { existsSync, readFileSync } from 'node:fs';
import { resolve } from 'node:path';

export const readFrameNames = (manifest) => {
	const repoAssetRoot = resolve(process.cwd(), '..', 'public', 'assets', 'phantasie', 'amiga');
	const assetRoot = existsSync(repoAssetRoot)
		? repoAssetRoot
		: resolve(process.cwd(), 'public', 'assets', 'phantasie', 'amiga');
	const data = JSON.parse(readFileSync(resolve(assetRoot, `${manifest}.json`), 'utf8'));
	return data.frames.map((frame) => frame.name);
};

export const titleCase = (raw) =>
	raw.toLowerCase().replace(/(^|[^a-z])([a-z])/g, (_match, sep, ch) => sep + ch.toUpperCase());

export const buildLabels = (names) =>
	new Map(names.map((name) => [name, { visible: titleCase(name.replace(/^\d+-/, '')) }]));

export const monsterLabels = buildLabels(readFrameNames('mstr1'));
export const raceLabels = buildLabels(readFrameNames('party1'));

Every monster and playable-race sprite extracted from Phantasie I's raw
`.pat` sprite banks — see [Graphics Formats](/phantasie/graphics/) for the
container format and addressing tables these were pulled from.

**Where the names came from.** The extraction pipeline already writes each
frame's index-prefixed monster/race name directly into the manifest (e.g.
`mstr1.json`'s `"0-SKELETON"`), read straight from the same 58-byte-slot
monster name table (`game+0x0C88`) and race name table (`game+0x69BE`) — see
`docs/phantasie/data-tables.md` §§1.1, 1.4 for the full offset evidence.
This page strips the numeric prefix and title-cases the remainder for
display — no name here is invented; every one traces back to the
executable's own string table.

Hover a card to see the raw `index-NAME` frame name and pixel dimensions.
Click a card (or press Enter/Space) to open it enlarged.

## Monsters

80 entries in each of `mstr1.pat` (normal pose) and `mstr2.pat` (attacking
pose) — the same 80 creatures, same index, same name, in each file. Summing
every entry's computed sprite size reproduces `mstr1.pat`'s exact
68,016-byte length with zero discrepancy (see
[Graphics Formats](/phantasie/graphics/)).

## A known colour defect

`pal3`, the palette these four atlases render under, has one still-uncorrected
colour index (12 of 16) — see [Graphics Formats](/phantasie/graphics/) §
"The palette correction." Any sprite using that index currently renders a
patch of pure red where the correct colour is a pale salmon. This is a
tracked, open item (`gfx-p1-palette-decode-variant`), not a rendering bug in
this gallery.
